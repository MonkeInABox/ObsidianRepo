### LiDAR Example
``` c
#include "eyebot.h"

int main() {
	int i, scan[360];
	while(1) {
		LCDMenu("SCAN", "", "", "END");
		if (KEYGet() == KEY4) return 0;
		LCDClear();
		LIDARClear();
		LIDARGet(scan);
		for (i = 0; i < 360; i++) {
			LCDLine(i, 250-scan[i]/10, i, 250, BLUE);
		}
		LCDLine(180, 0, 180, 250, RED); //straight
		LCDLine(90, 0, 90, 250, GREEN);
		LCDLine(270, 0, 270, 250, GREEN);
		LCDSetPrintf(19, 0, "              -90             0                       90");
	}
}
```
